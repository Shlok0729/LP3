def solve_n_queens(n):
    """
    Solve N-Queens for a given n.
    Returns a list of solutions; each solution is a list of strings,
    where 'Q' marks a queen and '.' marks an empty square.
    """
    solutions = []
    board = ["." * n for _ in range(n)]

    cols = set()       # occupied columns
    pos_diag = set()   # occupied positive diagonals (r + c)
    neg_diag = set()   # occupied negative diagonals (r - c)

    def place(row):
        if row == n:
            solutions.append(board.copy())
            return

        for col in range(n):
            if col in cols or (row + col) in pos_diag or (row - col) in neg_diag:
                continue

            # place queen
            cols.add(col)
            pos_diag.add(row + col)
            neg_diag.add(row - col)
            board[row] = board[row][:col] + "Q" + board[row][col + 1:]

            place(row + 1)

            # remove queen (backtrack)
            cols.remove(col)
            pos_diag.remove(row + col)
            neg_diag.remove(row - col)
            board[row] = board[row][:col] + "." + board[row][col + 1:]

    place(0)
    return solutions


def print_solutions(solutions):
    for idx, sol in enumerate(solutions, 1):
        print(f"\nSolution #{idx}:")
        for row in sol:
            print(row)
    print(f"\nTotal solutions: {len(solutions)}")


if __name__ == "__main__":
    n = int(input("Enter n (number of queens): "))
    sols = solve_n_queens(n)
    if sols:
        print_solutions(sols)
    else:
        print("No solutions found.")
